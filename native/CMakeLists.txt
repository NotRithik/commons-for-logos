cmake_minimum_required(VERSION 3.20)
project(CommonsNativeBasecampPlugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Core Gui Qml Quick RemoteObjects)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(LOGOS_DEPS_DIR "" CACHE PATH "Directory containing pinned logos-view-module, logos-plugin-qt and logos-protocol checkouts")
set(LOGOS_VIEW_GENERATOR "" CACHE FILEPATH "Path to the official logos-view-generator executable")
if(NOT IS_DIRECTORY "${LOGOS_DEPS_DIR}" OR NOT EXISTS "${LOGOS_VIEW_GENERATOR}")
  message(FATAL_ERROR "Set LOGOS_DEPS_DIR and LOGOS_VIEW_GENERATOR. See docs/NATIVE-BUILD.md.")
endif()
set(MOD "${REPO}/module")
set(SDK "${REPO}/sdk/src")
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
configure_file("${MOD}/metadata.json" "${GEN}/metadata.json" COPYONLY)
execute_process(COMMAND "${LOGOS_VIEW_GENERATOR}"
 --backend ui --metadata "${MOD}/metadata.json" --rep "${MOD}/src/commons_primitives_ui.rep"
 --backend-class CommonsPrimitivesBackend --backend-header commons_primitives_backend.h
 --output-dir "${GEN}" COMMAND_ERROR_IS_FATAL ANY)
set(LOGOS_REP_CLASS CommonsPrimitivesUi)
set(LOGOS_REP_BASE commons_primitives_ui)
set(LOGOS_FACTORY_CLASS CommonsPrimitivesUiReplicaFactory)
set(LOGOS_QML_URI Logos.CommonsPrimitives)
set(LOGOS_QML_TYPE_NAME CommonsPrimitives)
foreach(name LogosViewPluginBase.h LogosViewPluginBase.cpp LogosViewReplicaFactory.h LogosViewReplicaFactory.cpp)
 configure_file("${LOGOS_DEPS_DIR}/logos-view-module/cmake/${name}.in" "${GEN}/${name}" @ONLY)
endforeach()
# No declared module dependencies: an empty typed dependency aggregate.
# The real Basecamp host supplies LogosAPI; no mock API is created here.
file(WRITE "${GEN}/logos_sdk.h" "#pragma once\nclass LogosAPI;\nstruct LogosModules { explicit LogosModules(LogosAPI*) {} };\n")
set(SDK_SOURCES "${SDK}/CommonsLogosCliClient.h" "${SDK}/CommonsLogosCliClient.cpp" "${SDK}/CommonsLogosSchemas.h" "${SDK}/CommonsLogosSchemas.cpp")
add_library(commons_primitives_ui_plugin SHARED
 "${MOD}/src/commons_primitives_backend.h" "${MOD}/src/commons_primitives_backend.cpp"
 "${GEN}/commons_primitives_ui_ui_glue.h" "${GEN}/commons_primitives_ui_ui_glue.cpp"
 "${GEN}/LogosViewPluginBase.h" "${GEN}/LogosViewPluginBase.cpp" ${SDK_SOURCES})
set_target_properties(commons_primitives_ui_plugin PROPERTIES PREFIX "" BUILD_RPATH "${QT_BUNDLE_FRAMEWORKS}" INSTALL_RPATH "@executable_path/../Frameworks")
target_include_directories(commons_primitives_ui_plugin PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${MOD}/src" "${SDK}" "${GEN}" "${LOGOS_DEPS_DIR}/logos-view-module/cpp" "${LOGOS_DEPS_DIR}/logos-plugin-qt/core" "${LOGOS_DEPS_DIR}/logos-plugin-qt/cpp" "${LOGOS_DEPS_DIR}/logos-protocol/cpp")
target_link_libraries(commons_primitives_ui_plugin PRIVATE Qt6::Core Qt6::RemoteObjects Qt6::Qml)
qt6_add_repc_sources(commons_primitives_ui_plugin "${MOD}/src/commons_primitives_ui.rep")
if(APPLE)
 target_link_options(commons_primitives_ui_plugin PRIVATE -undefined dynamic_lookup)
endif()
add_library(commons_primitives_ui_replica_factory SHARED "${GEN}/LogosViewReplicaFactory.h" "${GEN}/LogosViewReplicaFactory.cpp")
set_target_properties(commons_primitives_ui_replica_factory PROPERTIES PREFIX "")
target_include_directories(commons_primitives_ui_replica_factory PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${GEN}")
target_link_libraries(commons_primitives_ui_replica_factory PRIVATE Qt6::Core Qt6::RemoteObjects Qt6::Qml)
qt6_add_repc_replicas(commons_primitives_ui_replica_factory "${MOD}/src/commons_primitives_ui.rep")
if(APPLE)
 enable_language(OBJCXX)
 foreach(target commons_primitives_ui_plugin commons_primitives_ui_replica_factory)
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/macos_text_input.mm")
  target_link_libraries(${target} PRIVATE "-framework Foundation")
  set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/macos_text_input.mm" PROPERTIES COMPILE_FLAGS "-fobjc-arc")
 endforeach()
endif()
install(TARGETS commons_primitives_ui_plugin commons_primitives_ui_replica_factory LIBRARY DESTINATION .)
install(FILES "${MOD}/metadata.json" DESTINATION .)
install(DIRECTORY "${MOD}/src/qml" DESTINATION .)
install(DIRECTORY "${MOD}/icons" DESTINATION .)

option(COMMONS_BUILD_TESTS "Build isolated native SDK tests" ON)
if(COMMONS_BUILD_TESTS)
  enable_testing()
  find_package(Qt6 REQUIRED COMPONENTS Test QuickTest)
  add_executable(commons_qml_controls_test "${MOD}/tests/qml_test_main.cpp")
  target_link_libraries(commons_qml_controls_test PRIVATE Qt6::QuickTest)
  target_compile_definitions(commons_qml_controls_test PRIVATE QUICK_TEST_SOURCE_DIR="${MOD}/tests/qml")
  add_test(NAME commons_qml_controls COMMAND commons_qml_controls_test)
  set_tests_properties(commons_qml_controls PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  add_executable(commons_test_fake_cli "${MOD}/tests/fake_commons_logos_cli.cpp")
  set_target_properties(commons_test_fake_cli PROPERTIES OUTPUT_NAME "commons-logos-cli" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-only")
  target_link_libraries(commons_test_fake_cli PRIVATE Qt6::Core)
  add_executable(commons_logos_cli_client_test "${MOD}/tests/commons_logos_cli_client_test.cpp" ${SDK_SOURCES})
  target_include_directories(commons_logos_cli_client_test PRIVATE "${SDK}")
  target_link_libraries(commons_logos_cli_client_test PRIVATE Qt6::Core Qt6::Test)
  target_compile_definitions(commons_logos_cli_client_test PRIVATE COMMONS_SDK_TESTING "COMMONS_TEST_FAKE_CLI=\"$<TARGET_FILE:commons_test_fake_cli>\"")
  add_dependencies(commons_logos_cli_client_test commons_test_fake_cli)
  add_test(NAME commons_native_sdk COMMAND commons_logos_cli_client_test)
  add_subdirectory(backend-tests)
  if(APPLE)
    add_executable(commons_macos_text_input_test "${MOD}/tests/macos_text_input_test.mm" "${CMAKE_CURRENT_SOURCE_DIR}/macos_text_input.mm")
    target_compile_options(commons_macos_text_input_test PRIVATE -fobjc-arc)
    target_link_libraries(commons_macos_text_input_test PRIVATE "-framework Foundation")
    add_test(NAME process_local_autofill_setting COMMAND commons_macos_text_input_test)
    set_tests_properties(process_local_autofill_setting PROPERTIES ENVIRONMENT "COMMONS_DISABLE_NATIVE_AUTOFILL=1")
  endif()
endif()
